# ROOT directory - ma poi non si riesce a disconnettere il device

# ROOT directory
net use S: \\sshfs.k\pi@192.168.1.31\..\..
# ROOT directory - chiede utenza e password
net use S: \\sshfs.r\pi@192.168.1.31

# Specifica directory
