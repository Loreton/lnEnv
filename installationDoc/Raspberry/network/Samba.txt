sudo apt-get update && sudo apt-get upgrade
sudo apt-get install samba samba-common-bin

Create directory to be shared
sudo mkdir -m 1777 /share


sudo vi /etc/samba/smb.conf

[PiShare]
   Comment = Loreto-PI Shared
   Path = /home/pi/LnShare
   Browseable = yes
   Writeable = Yes
   only guest = no
   create mask = 0777
   directory mask = 0777
   Public = yes
   Guest ok = yes

Create a user and start Samba (deve essere un utente definito al sistema)
sudo smbpasswd -a pi
psw: PiSamba

sudo systemctl restart smbd
